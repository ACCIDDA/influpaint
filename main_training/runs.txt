# Paper calibration commands; run from the research repository root.
# Run each stage after its prerequisites finish. This is a command reference.

cp influpaint-paper/influpaint_paper_reproduction_data/datasets/training/TS_*_2025-07-17.nc training_datasets/

sbatch main_training/generated/inpaint_array_paper-2025-07-22.run

python -m evaluation.prepare_dataset_for_scoringutils
